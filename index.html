    <div class="cell"><img id="img1"></div>

    <div class="cell"><img id="img2"></div>

    <div class="cell"><img id="img3"></div>

    <div class="cell"><img id="img4"></div>

  </div>


  <script>

    const TOTAL_SHAPES = 36; // adjust if needed

    const PATH = "shapes/";


    function getRandomShape() {

      const num = Math.floor(Math.random() * TOTAL_SHAPES) + 1;

      return `${PATH}shape${num}.svg?v=${Date.now()}`;

    }


    function startRandomizer(elementId, interval) {

      const el = document.getElementById(elementId);


      function update() {

        el.src = getRandomShape();

      }


      update(); // initial load

      setInterval(update, interval);

    }


    // Independent refresh speeds

    startRandomizer("img1", 1000);   // 1 sec

    startRandomizer("img2", 7500);   // 7.5 sec

    startRandomizer("img3", 5000);   // 5 sec

    startRandomizer("img4", 12000);  // 12 sec

  </script>
